-- 28. Show top 3 customers in each city using RANK() window function.
-- Rank customers by purchase value within each city.
SELECT
    city,
    customer_id,
    total_purchase,
    city_rank
FROM (
    SELECT
        c.city,
        c.customer_id,
        SUM(od.price * od.quantity) AS total_purchase,
        RANK() OVER (
            PARTITION BY c.city
            ORDER BY SUM(od.price * od.quantity) DESC
        ) AS city_rank
    FROM customers c
    JOIN orders o
        ON c.customer_id = o.customer_id
    JOIN order_details od
        ON o.order_id = od.order_id
    GROUP BY c.city, c.customer_id
) ranked_customers
WHERE city_rank <= 3
ORDER BY city, city_rank;

-- 30. Calculate average order value (AOV) per customer.
-- Total sales / number of orders for each customer.

SELECT
    customer_id,
    SUM(order_sales) / COUNT(order_id) AS avg_order_value
FROM (
    SELECT
        o.customer_id,
        o.order_id,
        SUM(od.price * od.quantity) AS order_sales
    FROM orders o
    JOIN order_details od
        ON o.order_id = od.order_id
    GROUP BY o.customer_id, o.order_id
) t
GROUP BY customer_id;

-- 32. Identify customers who returned after long gaps between orders.Use LAG() to compare difference between successive order dates.

SELECT
    customer_id,
    order_id,
    order_date,
    prev_order_date,
    DATEDIFF(order_date, prev_order_date) AS days_between_orders
FROM (
    SELECT
        customer_id,
        order_id,
        order_date,
        LAG(order_date) OVER (PARTITION BY customer_id ORDER BY order_date) AS prev_order_date
    FROM orders
) t
WHERE prev_order_date IS NOT NULL
  AND DATEDIFF(order_date, prev_order_date) > 30  -- >30 days gap, adjust as needed
ORDER BY customer_id, order_date;
